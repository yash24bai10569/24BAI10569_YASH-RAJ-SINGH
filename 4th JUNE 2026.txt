package mongo;

import com.mongodb.client.*; 
import com.mongodb.client.model.Filters; 
import com.mongodb.client.model.Updates; 
import org.bson.Document; 
import java.util.Arrays;
import java.util.List;
 
public class MongoCRUDExample { 
 
    public static void main(String[] args) { 
 
        // Connect to MongoDB 
        MongoClient mongoClient = MongoClients.create("mongodb://localhost:27017"); 
 
        // Access database 
        MongoDatabase database = mongoClient.getDatabase("CollegeDB"); 
 
        // Access collection 
        MongoCollection<Document> collection = database.getCollection("Students"); 
 
        // ========================== 
        // CREATE (Insert Many) 
        // ========================== 
        Document student1 = new Document("id", 101) 
                .append("name", "John") 
                .append("department", "CSE") 
                .append("cgpa", 8.7); 

        Document student2 = new Document("id", 102) 
                .append("name", "Alice") 
                .append("department", "CSE") 
                .append("cgpa", 8.9); 

        Document student3 = new Document("id", 103) 
                .append("name", "Bob") 
                .append("department", "ECE") 
                .append("cgpa", 7.5); 

        // Group them into a list
        List<Document> studentList = Arrays.asList(student1, student2, student3);

        // Insert multiple documents at once
        collection.insertMany(studentList); 
        System.out.println("Multiple Documents Inserted Successfully"); 
 
        // ========================== 
        // READ (Find All) 
        // ========================== 
        System.out.println("\nAll Student Records:"); 
        FindIterable<Document> students = collection.find(); 
 
        for (Document doc : students) { 
            System.out.println(doc.toJson()); 
        } 
 
        // ========================== 
        // UPDATE (Update Many - Inc) 
        // ========================== 
        // Update CGPA for ALL students in the "CSE" department
        collection.updateMany( 
                Filters.eq("department", "CSE"), 
                Updates.inc("cgpa", 0.5) // Increases everyone's CGPA in CSE by 0.5
        ); 
        System.out.println("\nMultiple Documents Updated Successfully (CSE department bumped)"); 

        // ========================== 
        // UPDATE (Unset a Field) 
        // ========================== 
        // Remove the "cgpa" field entirely for Bob (id: 103)
        collection.updateOne(
                Filters.eq("id", 103),
                Updates.unset("cgpa")
        );
        System.out.println("Successfully unset the 'cgpa' field for student ID 103");

        // ========================== 
        // UPDATE (Rename a Field) 
        // ========================== 
        // Rename the "department" field to "major" for ALL students
        collection.updateMany(
                new Document(), // An empty document acts as a filter that matches everything
                Updates.rename("department", "major")
        );
        System.out.println("Successfully renamed the 'department' field to 'major' for all records");
 
        // ========================== 
        // READ Updated Records 
        // ========================== 
        System.out.println("\nFinal Records (Notice the missing CGPA for Bob and 'major' instead of 'department'):"); 
        FindIterable<Document> finalStudents = collection.find(); 
 
        for (Document doc : finalStudents) { 
            System.out.println(doc.toJson()); 
        }
 
        // ========================== 
        // DELETE (Delete Many) 
        // ========================== 
        // Delete all students who belong to the "CSE" major 
        // Note: We use "major" instead of "department" here because we just renamed it!
        collection.deleteMany(Filters.eq("major", "CSE")); 
 
        System.out.println("\nMultiple Documents Deleted Successfully"); 
 
        // Close connection 
        mongoClient.close(); 
    } 
}

package mongo;

import com.mongodb.client.*;
import com.mongodb.client.model.Filters;
import com.mongodb.client.model.Updates;
import org.bson.Document;

import java.util.Arrays;
import java.util.List;

public class MongoCRUDExample {

    public static void main(String[] args) {

        // Connect to MongoDB
        MongoClient mongoClient = MongoClients.create("mongodb://localhost:27017");

        // Access database
        MongoDatabase database = mongoClient.getDatabase("CollegeDB");

        // Access collection
        MongoCollection<Document> collection = database.getCollection("Students");

        // ==========================
        // CREATE (Insert Many)
        // ==========================
        Document student1 = new Document("id", 101)
                .append("name", "John")
                .append("department", "CSE")
                .append("cgpa", 8.7);

        Document student2 = new Document("id", 102)
                .append("name", "Alice")
                .append("department", "CSE")
                .append("cgpa", 8.9);

        Document student3 = new Document("id", 103)
                .append("name", "Bob")
                .append("department", "ECE")
                .append("cgpa", 7.5);

        List<Document> studentList = Arrays.asList(student1, student2, student3);

        collection.insertMany(studentList);
        System.out.println("Multiple Documents Inserted Successfully");

        // ==========================
        // READ (Find All)
        // ==========================
        System.out.println("\nAll Student Records:");

        FindIterable<Document> students = collection.find();

        for (Document doc : students) {
            System.out.println(doc.toJson());
        }

        // ==========================
        // UPDATE (Increment CGPA)
        // ==========================
        collection.updateMany(
                Filters.eq("department", "CSE"),
                Updates.inc("cgpa", 0.5)
        );

        System.out.println("\nCSE Students' CGPA Updated Successfully");

        // ==========================
        // UPDATE (UNSET Field)
        // ==========================
        collection.updateOne(
                Filters.eq("id", 103),
                Updates.unset("cgpa")
        );

        System.out.println("CGPA field removed for student ID 103");

        // ==========================
        // UPDATE (Rename Field)
        // ==========================
        collection.updateMany(
                new Document(),
                Updates.rename("department", "major")
        );

        System.out.println("Field 'department' renamed to 'major'");

        // ==========================
        // READ Updated Records
        // ==========================
        System.out.println("\nFinal Records:");

        FindIterable<Document> finalStudents = collection.find();

        for (Document doc : finalStudents) {
            System.out.println(doc.toJson());
        }

        // ==========================
        // DELETE (Delete Many)
        // ==========================
        collection.deleteMany(
                Filters.eq("major", "CSE")
        );

        System.out.println("\nCSE Students Deleted Successfully");

        // Close connection
        mongoClient.close();
    }
}
